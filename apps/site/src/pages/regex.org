#+TITLE: Regex Without Backtracking
#+SUBTITLE: Compile patterns to a Thompson NFA and advance the whole active state set, one character per step

Every block here starts with a pattern between slashes (=/pattern/=, no flags), followed by one test input per line (=#= lines are comments). Unlike backtracking engines in [[file:javascript.org][JavaScript]] or [[file:python.org][Python]], this kernel compiles the pattern to a *Thompson NFA* and advances the entire set of active states simultaneously. Click *Step* to inspect *The machine* (compiled NFA states), *NFA states* (currently live states), *Input* (character caret and match brackets), *Captures*, and *console*.

* How a Thompson NFA runs

Instead of guessing one branch and backtracking on failure, a Thompson NFA keeps every reachable state alive at once through $epsilon$-transitions. After reading each character, duplicate states are merged into a set.

#+CAPTION: Thompson NFA for /(a|b)+c/ with epsilon split and loop-back transitions
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2": canvas, draw

#canvas(length: 1cm, {
  import draw: *
  let mono(body) = text(font: "DejaVu Sans Mono", size: 8pt, body)

  // 31pt-diameter states: widest label is "acc", leaving ~9pt on either side
  let r = 0.55cm

  let s0 = (0cm, 0cm)
  let a = (2.3cm, 1cm)
  let b = (2.3cm, -1cm)
  let s3 = (4.6cm, 0cm)
  let c = (7cm, 0cm)
  let acc = (9.4cm, 0cm)

  // point on the circle around `center` in the direction of `toward`
  let rim(center, toward) = {
    let dx = (toward.at(0) - center.at(0)) / 1cm
    let dy = (toward.at(1) - center.at(1)) / 1cm
    let l = calc.sqrt(dx * dx + dy * dy)
    (center.at(0) + dx / l * r, center.at(1) + dy / l * r)
  }

  let disc(pos, body, hi: false) = {
    circle(pos, radius: r,
      stroke: 0.9pt + if hi { rgb("#1A9988") } else { luma(120) },
      fill: if hi { rgb("#F2F8F6") } else { white })
    content(pos, body)
  }

  // one transition: rim to rim, symbol label parked to one side of the stroke
  let spoke(from, to, lbl, side: 1, color: luma(130)) = {
    let p = rim(from, to)
    let q = rim(to, from)
    let m = ((p.at(0) + q.at(0)) / 2, (p.at(1) + q.at(1)) / 2)
    let dx = (to.at(0) - from.at(0)) / 1cm
    let dy = (to.at(1) - from.at(1)) / 1cm
    let l = calc.sqrt(dx * dx + dy * dy)
    let off = 0.33cm * side
    line(p, q, stroke: 0.8pt + color, mark: (end: ">", fill: color, scale: 0.85))
    content((m.at(0) - dy / l * off, m.at(1) + dx / l * off), lbl)
  }

  disc(s0, [$s_0$], hi: true)
  disc(a, [$a$])
  disc(b, [$b$])
  disc(s3, [$s_3$])
  disc(c, [$c$])
  disc(acc, [acc], hi: true)

  spoke(s0, a, [$epsilon$])
  spoke(s0, b, [$epsilon$], side: -1)
  spoke(a, s3, mono("a"), color: rgb("#1A9988"))
  spoke(b, s3, mono("b"), side: -1, color: rgb("#1A9988"))
  spoke(s3, c, [$epsilon$])
  spoke(c, acc, mono("c"), color: rgb("#1A9988"))

  // the `+` loop back to the split runs in its own lane above every state
  let lane = 2.05cm
  line(
    (s3.at(0), r), (s3.at(0), lane), (s0.at(0), lane), (s0.at(0), r),
    stroke: 0.8pt + rgb("#B3402E"), radius: 0.14cm,
    mark: (end: ">", fill: rgb("#B3402E"), scale: 0.85),
  )
  content(
    ((s0.at(0) + s3.at(0)) / 2, lane + 0.38cm),
    [$epsilon$ (loop)],
  )
})
#+END_SRC

* Alternation, quantifiers, and capturing groups

Because both sides of =|= stay alive in the same state set, =/a|ab/= and =/ab|a/= always agree on the longest span (unlike JS/Python, where =/a|ab/= stops early at ="a"=). Counted quantifiers like ={2,3}= expand into repeated NFA states that you can see directly in *The machine*.

#+NAME: regex-alt-quant
#+BEGIN_SRC regex
/a|ab/
ab
a
xab
#+END_SRC

#+RESULTS:
: match [0,2)
: match [0,1)
: match [1,3)

Capturing groups =(...)= emit zero-width open and close marker states in the NFA. Step through =regex-groups= with *Captures* open to watch group 1 and group 2 record their half-open =[start, end)= spans.

#+NAME: regex-groups
#+BEGIN_SRC regex
/(a(b))c/
xabc
#+END_SRC

#+RESULTS:
: match [1,4) groups: 1=[1,3) 2=[2,3)

*Try editing:* change =/(a(b))c/= to =/(?:a(b))c/= (non-capturing outer group) and re-run — group 1 is now =[2,3)=.

* Why =/(a*)*/= stays linear ($O(n dot.c |Q|)$)

A backtracking engine can take $O(2^n)$ steps on nested quantifiers like =/(a*)*/= against =aaa...ab=, because it retries every way of splitting the =a=s. A Thompson NFA merges duplicate active states at every character, bounding total work to $O(n dot.c |Q|)$.
