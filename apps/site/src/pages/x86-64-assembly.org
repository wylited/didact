* Linux syscalls and the register ABI

On Linux x86-64, user code asks the kernel to perform I/O through =syscall=. Before executing =syscall=, you place the service number in =rax= and up to three arguments in =rdi=, =rsi=, and =rdx=. When the kernel returns, the result replaces =rax=, while =rip= advances past the 2-byte =0f 05= instruction.

Step through =hello= below. Watch =rax=, =rdi=, =rsi=, and =rdx= populate one by one in *Registers*, then step across =syscall=: *Output* prints the string and =rax= becomes =14= (the byte length computed at assembly time by =msg_len equ $ - msg=).

#+NAME: hello
#+BEGIN_SRC asm
; hello.asm — Linux x86-64 write + exit syscalls
section .text
start:
    mov rax, 1            ; sys_write
    mov rdi, 1            ; fd 1 = stdout
    mov rsi, msg          ; rsi <- address of msg in .data (0x601000)
    mov rdx, msg_len      ; rdx <- 14 bytes (including '\n')
    syscall               ; write(1, msg, 14) -> rax = 14

    mov rax, 60           ; sys_exit
    xor rdi, rdi          ; exit code 0 (shorter encoding than mov rdi, 0)
    syscall               ; halts the emulator

section .data
msg:     db "Hello, world!", 10    ; 10 = ASCII '\n'
msg_len  equ $ - msg
#+END_SRC

#+RESULTS:
: Hello, world!

*Try editing:* replace =mov rdx, msg_len= with =mov rdx, 5= and hit *Run* — *Output* truncates to =Hello= and =rax= returns =5=, even though all 14 bytes still sit untouched in *.data*. Check *Disassembly* too: notice =xor rdi, rdi= encodes in 3 bytes (=48 31 ff=), shorter than =mov rdi, 0=.

* Flags, loops, and turning integers into ASCII

Unlike [[file:riscv.org][RISC-V]] or [[file:mips.org][MIPS]], x86-64 splits comparison from branching through *EFLAGS*. =cmp rbx, 10= subtracts $10$ from =rbx=, discards the numeric difference, and records zero (=ZF=) and borrow (=CF=). The following =jbe next= jumps whenever =CF= or =ZF= is set (unsigned $"rbx" <= 10$).

To print the sum $sum_(i=1)^10 i = 55$ without libc, =div rcx= divides the 128-bit pair =rdx:rax= by =rcx=, placing the quotient in =rax= and the remainder in =rdx=. Adding ='0'= (=0x30=) converts each digit into ASCII, and two stores overwrite =digits= in *.data*.

#+NAME: sum-loop
#+BEGIN_SRC asm
; sum.asm — sum 1..10 = 55, convert to ASCII, and write to stdout
section .text
start:
    xor rax, rax          ; sum = 0
    mov rbx, 1            ; i = 1
next:
    add rax, rbx          ; sum += i
    inc rbx               ; i++
    cmp rbx, 10           ; set ZF/CF from (i - 10)
    jbe next              ; loop while i <= 10

    mov rcx, 10           ; divisor = 10
    xor rdx, rdx          ; clear high 64 bits of rdx:rax before div
    div rcx               ; rax = 5 (tens), rdx = 5 (ones)
    add al, '0'           ; 5 -> 0x35 ('5')
    add dl, '0'
    mov [digits], al      ; overwrite .data[0]
    mov [digits+1], dl    ; overwrite .data[1]
    mov byte [digits+2], 10

    mov rax, 1            ; write(1, digits, 3)
    mov rdi, 1
    mov rsi, digits
    mov rdx, 3
    syscall

    mov rax, 60           ; exit(0)
    xor rdi, rdi
    syscall

section .data
digits:  db "00", 10      ; starts as 30 30 0a; becomes 35 35 0a at runtime
#+END_SRC

#+RESULTS:
: 55

*Try editing:* inspect *.data* at =0x601000= after running — =30 30 0a= (="00\n"=) became =35 35 0a= (="55\n"=). Now hit *Reset* and watch *.data* snap back to =30 30 0a=. Then change =cmp rbx, 10= to =cmp rbx, 9= and re-run to see =45= (=34 35 0a=).

* The stack, =call=, and =ret=

=call square= decrements =rsp= by 8 bytes, writes the return address onto the stack, and jumps to =square=. =ret= pops that 8-byte address back into =rip=. Following the System V AMD64 ABI, the first integer argument arrives in =rdi= and the return value goes back in =rax=.

#+NAME: stack-call
#+BEGIN_SRC asm
; stack-call.asm — watch rsp move by 8 bytes across call / ret
section .text
start:
    mov rdi, 9            ; argument n = 9
    call square           ; pushes return address (rsp -= 8), jumps to square

    ; rax is now 81 -> split into two ASCII digits
    mov rcx, 10
    xor rdx, rdx
    div rcx               ; rax = 8, rdx = 1
    add al, '0'
    add dl, '0'
    mov [out], al
    mov [out+1], dl

    mov rax, 1
    mov rdi, 1
    mov rsi, out
    mov rdx, 3
    syscall

    mov rax, 60
    xor rdi, rdi
    syscall

square:
    mov rax, rdi          ; rax = n
    imul rax, rdi         ; rax = n * n
    ret                   ; pops return address into rip (rsp += 8)

section .data
out:     db "00", 10
#+END_SRC

#+RESULTS:
: 81

*Try editing:* step across =call square= and =ret= while watching =rsp= and =rip= in *Registers*. Change =mov rdi, 9= to =mov rdi, 7= to verify =49=, or add =push rbx= / =pop rbx= inside =square= and watch =rsp= dip by another 8 bytes.
