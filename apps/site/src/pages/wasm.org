#+SUBTITLE: The i32 core of WAT: operand stack, binary opcodes, structured branches, and linear memory

This page runs a pure-TypeScript WebAssembly Text (WAT) interpreter for the =i32= core. Click *Step* or set gutter breakpoints to watch *Instructions* (showing real binary WebAssembly opcode bytes like =0x41= and =0x6a=), *Operand stack*, *Globals*, *Call stack*, *Memory*, and *Output*. These are the same binary opcodes emitted by clang in [[file:c.org][C]] and [[file:cpp.org][C++]].

* Arithmetic on an operand stack

WebAssembly has no CPU registers. Every instruction pushes onto or pops from a typed operand stack: =(i32.add (i32.const 20) (i32.const 22))= is just folded notation for pushing =20=, pushing =22=, then executing =i32.add= (=0x6a=).

#+CAPTION: Operand stack transitions for (i32.add (i32.const 20) (i32.const 22))
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2": canvas, draw

#context {
  canvas(length: 1pt, {
    import draw: *
    let mono = "DejaVu Sans Mono"
    let accent = rgb("#1A9988")
    let soft = rgb("#F2F8F6")
    let cols = (
      (op: "start", items: ()),
      (op: "i32.const 20", items: ("20",)),
      (op: "i32.const 22", items: ("20", "22")),
      (op: "i32.add", items: ("42",)),
    )
    let capw = cols
      .map(c => measure(text(font: mono, size: 8pt)[#c.op]).width)
      .fold(0pt, calc.max)
    let cw = calc.max(capw + 20pt, 76pt)
    let gap = 26pt
    let step = cw + gap
    let cellh = 17pt
    let fh = 48pt
    for (i, c) in cols.enumerate() {
      let x = i * step
      rect((x, 0pt), (x + cw, fh), stroke: 0.7pt + luma(170), fill: rgb("#FAFAFA"))
      content((x + cw / 2, fh + 6pt), text(font: mono, size: 8pt, fill: luma(60))[#c.op], anchor: "south")
      for (j, v) in c.items.enumerate() {
        let hl = v == "42"
        let y = 5pt + j * (cellh + 2.5pt)
        rect((x + 10pt, y), (x + cw - 10pt, y + cellh),
          stroke: 0.8pt + if hl { accent } else { luma(110) },
          fill: if hl { soft } else { white })
        content((x + cw / 2, y + cellh / 2), text(font: mono, size: 9.5pt, weight: "bold")[#v])
      }
      if i < cols.len() - 1 {
        let y = fh / 2
        line((x + cw, y), (x + step, y),
          stroke: 0.9pt + accent, mark: (end: ">", fill: accent))
      }
    }
  })
}
#+END_SRC

Step through =stack-arith= below and watch *Operand stack* grow and shrink. Notice =i32= multiplication wraps modulo $2^(32)$ ($65536^2 equiv 0 (mod 2^(32))$), and signed division =i32.div_s= truncates toward zero ($-7 / 2 = -3$).

#+NAME: stack-arith
#+BEGIN_SRC wasm
(module
  (import "didact" "print" (func $print (param i32)))
  (func (export "main")
    (call $print (i32.add (i32.const 20) (i32.const 22)))
    (call $print (i32.mul (i32.const 65536) (i32.const 65536)))
    (call $print (i32.div_s (i32.const -7) (i32.const 2)))))
#+END_SRC

* Locals and recursive frames on *Call stack*

Parameters and =(local ...)= slots are zero-initialized per call frame. Set a breakpoint on line 6 below and press *Run* repeatedly: watch *Call stack* stack up frames for =$n = 10, 9, \dots, 1= and then unwind as each =i32.mul= finishes.

#+NAME: factorial
#+BEGIN_SRC wasm
(module
  (import "didact" "print" (func $print (param i32)))
  (func $fact (param $n i32) (result i32)
    (if (result i32) (i32.le_s (local.get $n) (i32.const 1))
      (then (i32.const 1))
      (else (i32.mul (local.get $n) (call $fact (i32.sub (local.get $n) (i32.const 1)))))))
  (func (export "main")
    (call $print (call $fact (i32.const 10)))))
#+END_SRC

#+RESULTS:
: 3628800

* Structured control flow: =block=, =loop=, and =br=

WebAssembly forbids arbitrary =goto= so bytecode can be validated in a single linear pass. Instead, =br $label= jumps to opposite ends depending on the container: targeting a =block= exits forward (like =break=), while targeting a =loop= jumps backward to the header (like =continue=).

#+CAPTION: Opposite branch directions for block (exit forward) vs loop (repeat backward)
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2": canvas, draw

#context canvas(length: 1pt, {
  import draw: *
  let accent = rgb("#1A9988")
  let red = rgb("#B3402E")
  let soft = rgb("#F2F8F6")
  let mono = "DejaVu Sans Mono"
  let blk(head, note) = stack(
    dir: ttb, spacing: 4pt,
    align(center, text(weight: "bold")[#head]),
    align(center, text(font: mono, size: 8pt)[#note]),
  )
  let loop-blk = blk([loop \$again], [check i > 5 -> br_if \$done])
  let body-blk = blk([body], [sum += i; i += 1; br \$again])
  let end-blk = blk([end of block \$done], [call \$print(sum)])
  let wl = calc.max(measure(loop-blk).width, measure(body-blk).width) + 24pt
  let wr = measure(end-blk).width + 24pt
  let h = 40pt
  let v = 40pt
  let g = 78pt
  let top = h + v
  // body (top of the loop body)
  rect((0pt, 0pt), (wl, h), stroke: 0.8pt + luma(130), fill: white)
  content((wl / 2, h / 2), body-blk)
  // loop header
  rect((0pt, top), (wl, top + h), stroke: 0.8pt + accent, fill: soft)
  content((wl / 2, top + h / 2), loop-blk)
  // exit target
  rect((wl + g, top), (wl + g + wr, top + h), stroke: 0.8pt + luma(130), fill: white)
  content((wl + g + wr / 2, top + h / 2), end-blk)
  // fall-through lane, left of centre
  let fx = wl / 2 - 20pt
  line((fx, top), (fx, h), stroke: 0.9pt + luma(120), mark: (end: ">", fill: luma(120)))
  // repeat lane, right of centre, labelled beside the arrow
  let bx = wl / 2 + 20pt
  line((bx, h), (bx, top), stroke: 1pt + accent, mark: (end: ">", fill: accent))
  content((bx + 6pt, h + v / 2), text(size: 8pt, fill: luma(50))[br \$again], anchor: "west")
  // forward exit lane to the block end
  let ey = top + h / 2
  line((wl, ey), (wl + g, ey), stroke: 1pt + red, mark: (end: ">", fill: red))
  content((wl + g / 2, ey + 7pt), text(size: 8pt, fill: luma(50))[br_if \$done], anchor: "south")
})
#+END_SRC

#+NAME: counted-loop
#+BEGIN_SRC wasm
(module
  (import "didact" "print" (func $print (param i32)))
  (func (export "main")
    (local $i i32)
    (local $sum i32)
    (block $done
      (loop $again
        (br_if $done (i32.gt_s (local.get $i) (i32.const 5)))
        (local.set $sum (i32.add (local.get $sum) (local.get $i)))
        (local.set $i (i32.add (local.get $i) (i32.const 1)))
        (br $again)))
    (call $print (local.get $sum))))
#+END_SRC

#+RESULTS:
: 15

* Linear memory is a flat byte array

=(memory 1)= allocates one 64 KiB page of zero-initialized bytes. Multi-byte stores like =i32.store= write little-endian bytes, while =i32.load8_u= and =i32.load8_s= read a single byte unsigned or sign-extended (just like =lbu= and =lb= in [[file:riscv.org][RISC-V]]).

#+NAME: memory-bytes
#+BEGIN_SRC wasm
(module
  (import "didact" "print" (func $print (param i32)))
  (import "didact" "print_char" (func $print_char (param i32)))
  (memory 1)
  (func (export "main")
    (i32.store8 (i32.const 0) (i32.const 0x2A))
    (call $print_char (i32.load8_u (i32.const 0)))
    (call $print (i32.load8_u (i32.const 0)))))
#+END_SRC

#+RESULTS:
: *42

*Try editing:* replace =0x2A= with =0xFF= and compare =i32.load8_u= (=255=) against =i32.load8_s= (=-1=) while watching byte =ff= appear at =0x0000= in *Memory*.
