#+TITLE: SQL, Row by Row
#+SUBTITLE: Watch table scans, nested-loop joins, grouping, sorting, and NULL logic execute one row at a time

SQL is declarative — you write *what* rows you want rather than writing a loop — yet the database still runs a pipeline of iterators underneath. Every *Step* on this page executes one row operation and updates four live panels: *Plan* (each operator's =seen / passed / emitted= counters), *Current row*, *Tables*, and *Results*.

* Pipelines: Scan -> Filter -> Project

When you step through a =SELECT ... WHERE= query, rows flow upward from the table scan through the filter predicate to the projection. Watch *Plan* as =Scan users= reads all 5 rows (=5/5/5=), ~Filter age >= 30~ passes 2 (=5/2/2=), and =Project= emits 2 (=2/2/2=).

#+CAPTION: Row-flow operator pipeline with seen / passed / emitted counts
#+BEGIN_SRC typst :results svg

* Nested-loop joins and $O(N dot.c M)$ cost

This engine executes =JOIN= as a plain nested loop so the quadratic probe cost is visible on every step. For each of the $N = 3$ rows in =users=, it scans all $M = 3$ rows in =orders=, testing $3 times 3 = 9$ pairs and emitting the 2 that match ~users.id = orders.user_id~.

#+CAPTION: Nested-loop join probing all 3 x 3 = 9 (user, order) pairs

* =GROUP BY=, =HAVING=, and the grouping rule

=WHERE= filters raw rows *before* grouping; =GROUP BY= folds rows into per-key buckets one step at a time while updating =COUNT=, =SUM=, =AVG=, =MIN=, and =MAX=; =HAVING= filters the finished groups afterward.

#+NAME: group-by
#+BEGIN_SRC sql
CREATE TABLE users (id INTEGER, name TEXT, city TEXT, age INTEGER);
INSERT INTO users VALUES
  (1, 'alice', 'HK', 34),
  (2, 'Bob', 'HK', 27),
  (3, 'carol', 'SF', 41),
  (4, 'dave', 'SF', 19);
SELECT city, COUNT(*), AVG(age) FROM users GROUP BY city;
SELECT city, COUNT(*) FROM users GROUP BY city HAVING COUNT(*) > 1 AND AVG(age) > 30;
#+END_SRC

#+RESULTS:
: INSERT 0 4
: 
:  city | COUNT(*) | AVG(age) 
: ------+----------+----------
:  HK   | 2        | 30.5     
:  SF   | 2        | 30.0     
: 
:  city | COUNT(*) 
: ------+----------
:  HK   | 2        

*Try editing:* change =SELECT city, COUNT(*)= on line 7 to =SELECT name, COUNT(*)= and run — SQL rejects the query before scanning a single row because =name= is neither grouped nor aggregated. Wrap it as =MIN(name)= to fix it.

* Sorting, =LIMIT=, and =LIKE=

=ORDER BY= runs a stable merge sort where each comparison is one step (numbers numerically, text in ASCII order with uppercase before lowercase, =NULL= last), and =LIMIT= halts the pipeline early once enough rows have been emitted.

#+NAME: order-limit-like
#+BEGIN_SRC sql
CREATE TABLE scores (player TEXT, points INTEGER);
INSERT INTO scores VALUES ('alice', 7), ('Bob', 12), ('carol', 12), ('dave', 3);
SELECT player, points FROM scores ORDER BY points DESC, player LIMIT 3;
SELECT player FROM scores WHERE player LIKE '%a%';
#+END_SRC

#+RESULTS:
: INSERT 0 4
: 
:  player | points 
: --------+--------
:  Bob    | 12     
:  carol  | 12     
:  alice  | 7      
: 
:  player 
: --------
:  alice  
:  carol  
:  dave   

* Exercise: high-value orders by city

Write a query over =users= and =orders= that joins them, groups by =users.city=, and keeps only cities whose total spend =SUM(orders.total)= is at least =100=. Step through *Plan* to verify the operator order: =Scan= -> =NestedLoop= -> =Group= -> =Having= -> =Project=.

#+NAME: ex-city-spend
#+EXERCISE: Join users and orders, group by city, and filter HAVING SUM(orders.total) >= 100
#+BEGIN_SRC sql
CREATE TABLE users (id INTEGER, name TEXT, city TEXT);
INSERT INTO users VALUES (1, 'alice', 'HK'), (2, 'Bob', 'HK'), (3, 'carol', 'SF');
CREATE TABLE orders (id INTEGER, user_id INTEGER, total INTEGER);
INSERT INTO orders VALUES (10, 1, 70), (11, 2, 50), (12, 3, 60);

  FROM users JOIN orders ON users.id = orders.user_id;
#+END_SRC

#+BEGIN_HINT
Join on ~users.id = orders.user_id~, add =GROUP BY users.city=, and filter groups with ~HAVING SUM(orders.total) >= 100~.
#+END_HINT

#+BEGIN_SOLUTION
#+BEGIN_SRC sql
CREATE TABLE users (id INTEGER, name TEXT, city TEXT);
INSERT INTO users VALUES (1, 'alice', 'HK'), (2, 'Bob', 'HK'), (3, 'carol', 'SF');
CREATE TABLE orders (id INTEGER, user_id INTEGER, total INTEGER);
INSERT INTO orders VALUES (10, 1, 70), (11, 2, 50), (12, 3, 60);
SELECT users.city, SUM(orders.total)
  FROM users JOIN orders ON users.id = orders.user_id
  GROUP BY users.city
  HAVING SUM(orders.total) >= 100;
#+END_SRC
