#+TITLE: JavaScript, Stepped Live
#+SUBTITLE: Type coercion, var vs let closures, hoisting/TDZ, and the event loop on a steppable JS interpreter

This page runs a hand-written JavaScript interpreter instead of your browser's opaque engine so you can single-step every statement, drop gutter breakpoints, and watch *Scopes*, *Call stack*, *Event loop*, and *console* update live.

* Abstract equality (~==~) vs strict equality (~===~)

~===~ checks type and value without conversion. ~==~ coerces operands step by step until their types match, which is how ~[] == false~ evaluates to =true=: =[]= coerces to =''=, =''= coerces to =0=, and =false= coerces to =0=.

| Expression          | Coercion steps                      | Result |
|---------------------+-------------------------------------+--------|
| ~'' == 0~           | =''= -> =0=                         | true   |
| ~'1' == 1~          | ='1'= -> =1=                        | true   |
| ~[] == false~       | =[]= -> =''= -> =0=, =false= -> =0= | true   |
| ~null == undefined~ | spec special case                   | true   |
| ~NaN == NaN~        | =NaN= never equals itself           | false  |
| ~[] === false~      | different types, no coercion        | false  |

#+NAME: coercion
#+BEGIN_SRC javascript
console.log('' == 0, '1' == 1);
console.log([] == false, [] == 0, [7] == 7);
console.log(null == undefined, null == 0, undefined == 0);
console.log(NaN == NaN, NaN === NaN);
console.log([] === false, '1' === 1);
#+END_SRC

#+RESULTS:
: true true
: true true true
: true false false
: false false
: false false

* Closures capture bindings, not snapshots

A closure holds a reference to the environment slot where a variable lives, not a copy of its value. A ~for (var i = ...)~ loop allocates *one* shared =i= slot for the entire enclosing scope, whereas ~for (let j = ...)~ allocates a fresh =j= slot on every iteration.

#+CAPTION: Shared var i slot versus per-iteration let j slots captured by loop closures
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2": canvas, draw

#let mono = "DejaVu Sans Mono"
#let accent = rgb("#1A9988")
#let accent-fill = rgb("#F2F8F6")
#let warn = rgb("#C2553F")
#let warn-fill = rgb("#FDF3F1")
#let fn-text(t) = text(font: mono, size: 8pt)[#t]
#let slot-text(t) = text(font: mono, size: 9pt, weight: "bold")[#t]
#let arrow-mark(c) = (end: ">", fill: c, length: 4pt, width: 3.6pt)

// Boxes are sized from measured labels, so nothing clips or wraps; the shared
// var cell spans all three rows, and every arrow is a straight horizontal run
// whose head lands on a cell border.
#context {
  let var-labels = ("withVar[0]", "withVar[1]", "withVar[2]")
  let let-labels = ("withLet[0]", "withLet[1]", "withLet[2]")
  let slot-labels = ("let j = 0", "let j = 1", "let j = 2")
  let fn-w = (var-labels + let-labels)
    .map(l => measure(fn-text(l)).width)
    .fold(0pt, calc.max) + 14pt
  let slot-w = slot-labels
    .map(l => measure(slot-text(l)).width)
    .fold(0pt, calc.max) + 16pt

  let h = 21pt
  let row-gap = 8pt
  let pitch = h + row-gap
  let stack-h = 3 * h + 2 * row-gap
  let arrow-gap = 26pt
  let group-gap = 42pt

  let x-a = 0pt
  let x-a-r = x-a + fn-w
  let x-var-l = x-a-r + arrow-gap
  let x-var-r = x-var-l + slot-w
  let x-b = x-var-r + group-gap
  let x-b-r = x-b + fn-w
  let x-let-l = x-b-r + arrow-gap
  let x-let-r = x-let-l + slot-w
  let cy(i) = -(i * pitch + h / 2)

  canvas(length: 1pt, {
    import draw: *
    for (i, l) in var-labels.enumerate() {
      rect((x-a, cy(i) - h / 2), (x-a-r, cy(i) + h / 2),
        stroke: 0.8pt + luma(130), fill: white)
      content(((x-a + x-a-r) / 2, cy(i)), fn-text(l))
      line((x-a-r, cy(i)), (x-var-l, cy(i)), stroke: 0.8pt + warn, mark: arrow-mark(warn))
    }
    rect((x-var-l, cy(1) - stack-h / 2), (x-var-r, cy(1) + stack-h / 2),
      stroke: 0.9pt + warn, fill: warn-fill)
    content(((x-var-l + x-var-r) / 2, cy(1)), slot-text("var i = 3"))

    for (i, l) in let-labels.enumerate() {
      rect((x-b, cy(i) - h / 2), (x-b-r, cy(i) + h / 2),
        stroke: 0.8pt + luma(130), fill: white)
      content(((x-b + x-b-r) / 2, cy(i)), fn-text(l))
      rect((x-let-l, cy(i) - h / 2), (x-let-r, cy(i) + h / 2),
        stroke: 0.9pt + accent, fill: accent-fill)
      content(((x-let-l + x-let-r) / 2, cy(i)), slot-text(slot-labels.at(i)))
      line((x-b-r, cy(i)), (x-let-l, cy(i)), stroke: 0.8pt + accent, mark: arrow-mark(accent))
    }
  })
}
#+END_SRC

* Hoisting and the temporal dead zone

Before line 1 runs, the interpreter registers every declaration in the scope. =var= starts out =undefined= and =function= declarations are immediately callable. =let= and =const= are registered too, but stay in the *temporal dead zone* until their declaration executes — reading =later= on line 5 throws before line 6 runs.

#+NAME: hoisting-tdz
#+BEGIN_SRC javascript
console.log(hoistedVar, typeof hoistedFn);
var hoistedVar = 'now set';
function hoistedFn() { return 'callable before its line'; }
console.log(hoistedVar, hoistedFn());
console.log(later);
let later = 'never reached';
#+END_SRC

#+RESULTS:
: undefined function
: now set callable before its line

*Try editing:* step once from line 1 and look at *Scopes* — =hoistedVar= is already listed as =undefined= before line 2 executes. Now move ~let later = 'now safe';~ above line 5 and re-run to eliminate the TDZ error.

* Event loop order: sync -> microtasks -> macrotasks

Synchronous top-level code runs to completion first. Next, the engine drains the *entire* microtask queue (=Promise.resolve().then(...)=) before taking a single macrotask (=setTimeout(..., 0)=).

#+CAPTION: Event loop priority: synchronous stack, then full microtask drain, then next macrotask
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2": canvas, draw

#let accent = rgb("#1A9988")
#let accent-fill = rgb("#F2F8F6")
#let lanes = (
  (title: "1. Call stack", sub: "sync code to empty", hl: false),
  (title: "2. Microtask queue", sub: "drain ALL Promise.then", hl: true),
  (title: "3. Macrotask queue", sub: "run ONE setTimeout", hl: false),
)

// Lanes are measured from their longest line, so the captions keep a real inset
// from the border; the ordering arrows run in the gaps only.
#context {
  let lane-w = lanes
    .map(l => calc.max(
      measure(text(weight: "bold", size: 9pt)[#l.title]).width,
      measure(text(size: 8pt)[#l.sub]).width,
    ))
    .fold(0pt, calc.max) + 26pt
  let lane-h = 38pt
  let lane-gap = 30pt
  let x0(i) = i * (lane-w + lane-gap)
  let y-top = 0pt
  let y-bot = y-top - lane-h
  let y-run = y-bot - 14pt

  canvas(length: 1pt, {
    import draw: *
    for (i, l) in lanes.enumerate() {
      rect((x0(i), y-bot), (x0(i) + lane-w, y-top),
        stroke: 0.9pt + if l.hl { accent } else { luma(130) },
        fill: if l.hl { accent-fill } else { white })
      content((x0(i) + lane-w / 2, y-top - 12pt), text(weight: "bold")[#l.title])
      content((x0(i) + lane-w / 2, y-top - 27pt), text(size: 8pt)[#l.sub])
      if i < 2 {
        line((x0(i) + lane-w, y-top - lane-h / 2), (x0(i + 1), y-top - lane-h / 2),
          stroke: 0.9pt + accent, mark: (end: ">", fill: accent, length: 4.2pt, width: 3.6pt))
      }
    }
    // once a macrotask has run, the next drain starts at the microtask lane again
    line((x0(2) + lane-w / 2, y-bot), (x0(2) + lane-w / 2, y-run),
      (x0(1) + lane-w / 2, y-run), (x0(1) + lane-w / 2, y-bot),
      stroke: 0.7pt + luma(130), mark: (end: ">", fill: luma(130), length: 4pt, width: 3.6pt))
    content(((x0(1) + x0(2)) / 2 + lane-w / 2, y-run - 5pt),
      text(size: 8pt)[drain again], anchor: "north")
  })
}
#+END_SRC

* Exercise: filter evens, double, and join

Fix the =filter= callback so it keeps only even numbers, doubles them, and prints =4 · 8 · 12=. Drop a breakpoint inside the callback to watch *Call stack* push a frame per element.

#+NAME: ex-filter-map
#+EXERCISE: Keep even numbers, double them, and print 4 · 8 · 12
#+CHECK: 4 · 8 · 12
#+BEGIN_SRC javascript
var numbers = [1, 2, 3, 4, 5, 6];
var evens = numbers.filter(function (n) { return n % 2 !== 0; });
var doubled = evens.map(function (n) { return n * 2; });
console.log(doubled.join(' · '));
#+END_SRC

#+BEGIN_HINT
Change ~n % 2 !== 0~ to ~n % 2 === 0~.
#+END_HINT

#+BEGIN_SOLUTION

var numbers = [1, 2, 3, 4, 5, 6];
var evens = numbers.filter(function (n) { return n % 2 === 0; });

console.log(doubled.join(' · '));
#+END_SRC
#+END_SOLUTION
