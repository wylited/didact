#+TITLE: Prolog, Search Exposed
#+SUBTITLE: Unification, SLD resolution, choice points, backtracking, and cut on a steppable Prolog engine

In Prolog you declare *facts* and *rules*, then pose a query (=?- ...=). Rather than executing statements top to bottom, the engine searches for variable bindings that satisfy every goal using *SLD resolution* and depth-first backtracking. Click *Step* or set gutter breakpoints to inspect *Goal stack*, *Choice points*, *Search* (the exact transition on the last step: =resolve=, =mismatch=, =backtrack=, =cut=, =solution=), *Bindings*, and *Output*.

* Facts, unification, and backtracking

=likes(ada, prolog).= states a fact; =?- likes(ada, X).= asks the unifier to bind the capitalized variable =X=. Run =facts-and-queries= once to get ~X = prolog~, then click *Run* again to trigger backtracking to ~X = lisp~, and a third time to exhaust the search (=No.=).

#+NAME: facts-and-queries
#+BEGIN_SRC prolog
likes(ada, prolog).
likes(ada, lisp).
likes(bob, prolog).
?- likes(ada, X).
#+END_SRC

#+RESULTS:
: Yes.
: X = prolog

* Rules, the *Goal stack*, and SLD search

A rule =grandparent(X, Z) :- parent(X, Y), parent(Y, Z).= rewrites one goal into two subgoals. Prolog always resolves the leftmost goal first, pushing a choice point whenever multiple clauses could match.

#+CAPTION: SLD resolution tree for grandparent(tom, G) showing choice points and backtracking
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2" as cetz

#let mono(body) = text(font: "DejaVu Sans Mono", size: 8pt, body)
#let pad = 7pt
#let q1 = mono[?- grandparent(tom, G)]
#let q2 = mono[parent(tom, Y), parent(Y, G)]
#let q3 = mono[parent(mary, G)  (Y = mary)]
#let solved = [*G = ann* #linebreak() #text(size: 8pt)[1st solution]]
#let choice = [*G = pat* #linebreak() #text(size: 8pt)[via choice point]]
#let teal = rgb("#1A9988")
#let red = rgb("#C2553F")

#context {
  let maxw(bodies) = calc.max(..bodies.map(body => measure(body).width))
  let maxh(bodies) = calc.max(..bodies.map(body => measure(body).height))
  // boxes are sized from the measured text, so no goal can wrap or clip
  let W = maxw((q1, q2, q3)) + 2 * pad
  let H = maxh((q1, q2, q3)) + 2 * pad
  let WO = maxw((solved, choice)) + 2 * pad
  let HO = maxh((solved, choice)) + 2 * pad
  let vgap = 20pt
  let bgap = 44pt
  let hgap = 22pt
  let y1 = H / 2
  let y2 = y1 - H - vgap
  let y3 = y2 - H - vgap
  let y4 = y3 - H / 2 - bgap - HO / 2
  let X = W / 2 + hgap + WO / 2
  let ytop = y3 - H / 2

  cetz.canvas(length: 1pt, {
    import cetz.draw: *
    let bx(name, pos, w, h, body, stroke: luma(130), fill: white) = content(
      pos, box(width: w - 2 * pad, height: h - 2 * pad, align(center + horizon, body)),
      name: name, frame: "rect", padding: pad, stroke: stroke, fill: fill,
    )
    bx("q1", (0, y1), W, H, q1, stroke: 0.8pt + luma(130))
    bx("q2", (0, y2), W, H, q2, stroke: 0.8pt + luma(130))
    bx("q3", (0, y3), W, H, q3, stroke: 0.8pt + luma(130))
    bx("ok", (-X, y4), WO, HO, solved, stroke: 0.8pt + teal, fill: rgb("#F2F8F6"))
    bx("co", (X, y4), WO, HO, choice, stroke: 0.8pt + red, fill: rgb("#FDF3F1"))
    let arrow(from, to, paint, weight: 0.8pt) = line(
      from, to, stroke: weight + paint, mark: (end: "stealth", fill: paint, scale: 0.7))
    arrow("q1.south", "q2.north", luma(120))
    arrow("q2.south", "q3.north", teal)
    arrow("q3.south", "ok.north", teal, weight: 0.9pt)
    arrow("q3.south", "co.north", red)
    // both tags share one baseline in the empty band above the branches; the lift
    // covers the drop of the slanted stroke across the wider tag, so ink stays clear
    let first = text(size: 8pt)[clause 2]
    let second = text(size: 8pt)[redo -> clause 3]
    let tag-y = ytop - bgap * 0.72 + 6pt + maxw((first, second)) / 2 * (bgap / X)
    content((-X * 0.72, tag-y), first, anchor: "south")
    content((X * 0.72, tag-y), second, anchor: "south")
  })
}
#+END_SRC

* Relations run backwards: splitting lists with =append/3=

Because Prolog predicates are *relations* rather than one-way functions, =append/3= splits a list just as effortlessly as it concatenates two lists. Step through =append-generates= with *Choice points* open: every recursive call pushes an alternative onto *Choice points*, and =findall/3= drives backtracking through all three splits of =[1, 2]=.

#+NAME: append-generates
#+BEGIN_SRC prolog
append([], Ys, Ys).
append([X|Xs], Ys, [X|Zs]) :- append(Xs, Ys, Zs).
?- findall(p(Xs, Ys), append(Xs, Ys, [1, 2]), Ps).
#+END_SRC

#+RESULTS:
: Yes.
: Xs = Xs
: Ys = Ys
: Ps = [p([], [1, 2]), p([1], [2]), p([1, 2], [])]

*Try editing:* change =[1, 2]= to =[1, 2, 3]= and re-run — =Ps= now collects all 4 splits of the 3-element list.

* Pruning search with cut (=!=)

The cut operator =!= always succeeds and immediately discards every choice point created since the current predicate was entered. Contrast =max3= (which commits once ~X >= Y~ succeeds) with =maxu= (which lacks =!= and backtracks into the fallback clause too).

#+CAPTION: Cut (!) discards the choice point to clause 2 once X >= Y succeeds
#+BEGIN_SRC typst :results svg
#set page(margin: 12pt)
#set text(size: 9pt)
#import "@preview/cetz:0.5.2" as cetz

#let mono(body) = text(font: "DejaVu Sans Mono", size: 8pt, body)
#let pad = 7pt
#let query = mono[max3(5, 3, M)]
#let taken = [*clause 1: 5 >= 3, !* #linebreak() #text(size: 8pt)[commits -> M = 5]]
#let pruned = [#strike[*clause 2: pruned*] #linebreak() #text(size: 8pt)[choice point dropped]]
#let teal = rgb("#1A9988")
#let red = rgb("#C2553F")

#context {
  let maxw(bodies) = calc.max(..bodies.map(body => measure(body).width))
  let maxh(bodies) = calc.max(..bodies.map(body => measure(body).height))
  let WT = maxw((query,)) + 2 * pad
  let HT = maxh((query,)) + 2 * pad
  let WB = maxw((taken, pruned)) + 2 * pad
  let HB = maxh((taken, pruned)) + 2 * pad
  let bgap = 46pt
  let hgap = 110pt
  let X = WB / 2 + hgap / 2
  let ytop = -HT / 2
  let ybot = ytop - bgap - HB / 2

  cetz.canvas(length: 1pt, {
    import cetz.draw: *
    let bx(name, pos, w, h, body, stroke: luma(130), fill: white) = content(
      pos, box(width: w - 2 * pad, height: h - 2 * pad, align(center + horizon, body)),
      name: name, frame: "rect", padding: pad, stroke: stroke, fill: fill,
    )
    bx("q", (0, 0), WT, HT, query, stroke: 0.8pt + luma(130))
    bx("c1", (-X, ybot), WB, HB, taken, stroke: 0.8pt + teal, fill: rgb("#F2F8F6"))
    // the discarded clause is faded, struck through and framed with a dashed rule
    bx("c2", (X, ybot), WB, HB, {
      set text(fill: luma(118))
      pruned
    }, stroke: (paint: luma(150), thickness: 0.8pt, dash: "dashed"), fill: luma(247))
    line("q.south", "c1.north", stroke: 0.9pt + teal,
      mark: (end: "stealth", fill: teal, scale: 0.7))
    line("q.south", "c2.north", stroke: (paint: red, thickness: 0.8pt, dash: "dashed"),
      mark: (end: "stealth", fill: red, scale: 0.7))
    let cut = text(size: 8pt)[cut!]
    let lift = 6pt + measure(cut).width / 2 * (bgap / X)
    content((X * 0.72, ytop - bgap * 0.72 + lift), cut, anchor: "south")
  })
}
#+END_SRC

#+BEGIN_SRC prolog

max3(_, Y, Y).
maxu(X, Y, X) :- X >= Y.
maxu(_, Y, Y).
